"use client";

import { useState } from "react";
import { ShoppingCart } from "lucide-react";
import { useTranslations } from "next-intl";

import { useCartContext } from "@/features/cart/context/cart-provider";
import { TOAST_STATE } from "@/shared/lib/constants/toast-states.constant";
import { Button } from "@/shared/ui/button";
import { addToast } from "@/shared/ui/toast";
import type { IProductOverviewResponse } from "@/features/products/types/product";
import type { IApiResponse } from "@/shared/types/api";
import { cn } from "cn";

interface IAddToCartButtonProps {
  productId: string;
  stock: number;
  quantity?: number;
  variant?: "icon" | "full";
  isPopular?: boolean;
  className?: string;
  disabled?: boolean;
  onError?: (error: unknown) => void;
}

export default function AddToCartButton({
  productId,
  quantity = 1,
  variant = "icon",
  isPopular = false,
  className,
  stock,
  disabled = false,
  onError,
}: IAddToCartButtonProps) {
  const t = useTranslations("Cart");

  const { addToCart, getProductQuantity } = useCartContext();
  const [isAdding, setIsAdding] = useState(false);

  const productQuantity = getProductQuantity(productId);
  const isOutOfStock = stock <= 0;
  const isDisabled = disabled || isOutOfStock || isAdding;

  async function handleAddToCart() {
    if (isDisabled) return;

    setIsAdding(true);

    try {
      const response = await fetch(
        `/api/products/${encodeURIComponent(productId)}`,
        { cache: "no-store" },
      );
      const result: IApiResponse<IProductOverviewResponse> =
        await response.json();

      if (!response.ok || !result.status) {
        throw new Error(
          "message" in result ? result.message : "Product request failed",
        );
      }

      const product = result.payload.product;

      if (product.stock <= 0) {
        throw new Error("Product is out of stock");
      }

      await addToCart({
        product,
        quantity,
      });
    } catch (error) {
      console.error("Failed to add product to cart:", error);

      const message =
        error instanceof Error && error.message === "Product is out of stock"
          ? t("outOfStock")
          : t("addToCartFailed");

      addToast(TOAST_STATE.Unsuccessful, message);

      onError?.(error);
    } finally {
      setIsAdding(false);
    }
  }

  if (variant === "full") {
    return (
      <Button
        type="button"
        variant="primary"
        size="default"
        disabled={isDisabled}
        loading={isAdding}
        onClick={handleAddToCart}
        className={cn("relative", className)}
      >
        <span className="relative flex items-center justify-center">
          <ShoppingCart size={20} />

          {productQuantity > 0 && (
            <span className="absolute -top-0.5 left-1/2 -translate-x-1/2 -translate-y-1/2 text-[10px] font-bold">
              {productQuantity}
            </span>
          )}
        </span>

        {isAdding ? t("adding") : t("addToCart")}
      </Button>
    );
  }

  return (
    <Button
      type="button"
      variant="primary"
      disabled={isDisabled}
      loading={isAdding}
      onClick={handleAddToCart}
      className={cn(
        "bg-bg-maroon-secondary relative size-10.5 rounded-full",
        isPopular && "translate-y-[50%]",
        className,
      )}
    >
      <span className="relative flex items-center justify-center">
        <ShoppingCart size={20} />

        {productQuantity > 0 && (
          <span className="absolute -top-0.5 left-1/2 -translate-x-1/2 -translate-y-1/2 text-[10px] font-bold">
            {productQuantity}
          </span>
        )}
      </span>
    </Button>
  );
}

export function AuthLocaleToggle() {
  const locale = useLocale();
  const pathname = usePathname();
  const targetLocale = locale === "en" ? "ar" : "en";
  const label = locale === "en" ? "العربية" : "English";

      }`}
    >
      {label}
    </Link>
  );
}

===== src/shared/components/locale-switcher.tsx =====
"use client";

import { Suspense } from "react";
import { useSearchParams } from "next/navigation";
import { useLocale, useTranslations } from "next-intl";
import { Link, usePathname } from "@/i18n/navigation";
import { routing } from "@/i18n/routing";

const localeNames = { en: "English", ar: "العربية" };

/** Language links for the current path. `query` is kept so `page` and `limit` survive a locale change. */
function LocaleSwitcherLinks({ query }: { query: string }) {
  const locale = useLocale();
  const pathname = usePathname();
  const t = useTranslations();
  const params = new URLSearchParams(query);
  const search = Object.fromEntries(params.entries());

  const otherLocales = routing.locales.filter((l) => l !== locale);

  return (
    <nav aria-label={t("language")} className="flex gap-3 p-4">
      {otherLocales.map((nextLocale) => (
        <Link
          key={nextLocale}
          href={query ? { pathname, query: search } : pathname}
          locale={nextLocale}
          lang={nextLocale}
          dir={nextLocale === "ar" ? "rtl" : "ltr"}
          className={`dark:text-text-plain focus-visible:outline-ring rounded-sm px-2 py-1 text-zinc-700 underline-offset-4 hover:underline focus-visible:outline-2 ${nextLocale === "ar" ? "font-arabic" : "font-english"}`}
        >
          {localeNames[nextLocale]}
        </Link>
      ))}
    </nav>
  );
}

/** Reads the URL search string. Must render under Suspense or static rendering bails out. */
function LocaleSwitcherWithQuery() {
  const searchParams = useSearchParams();
  return <LocaleSwitcherLinks query={searchParams.toString()} />;
}

/** Locale links. The fallback omits the query until `useSearchParams` resolves. */
export function LocaleSwitcher() {
  return (
    <Suspense fallback={<LocaleSwitcherLinks query="" />}>
      <LocaleSwitcherWithQuery />
    </Suspense>
  );
}

  title: string;
  cover?: string | null;
  stock: number;
  rating?: number | null;
  price: number | string;
  discountType?: string | null;
  discountValue?: number | string | null;

  product: ProductCardProduct;
  variant?: "bestSelling" | "mostPopular";

            src={product.cover}
            alt={product.title}
            fill
            sizes={
              isPopular
                ? "(max-width: 640px) 50vw, (max-width: 1024px) 33vw, 25vw"
                : "302px"
            }
            className={cn(
              "object-cover",
              isPopular &&
                "transition-transform duration-300 group-hover:scale-105",

        {/* Product Badges */}

          className={cn(
            "pointer-events-none absolute top-2 z-20 flex flex-wrap gap-1",
            isPopular ? "inset-e-1 justify-end" : "right-2",
          )}
        >
          {product.stock <= 0 ? (
            <Badge
              variant="primary"
              className="bg-bg-danger text-text-secondary-fade dark:text-text-plain"
            >
              <span>{t("Product.outOfStock")}</span>
            </Badge>
          ) : (
            isPopular &&
            product.stock && (
              <>
                <Badge
                  variant="subtle"
                  className="bg-bg-muted text-text-zinc-700 dark:bg-bg-inverse dark:text-text-inverse"
                >
                  <span>{t("Product.new")}</span>

                  variant="secondary"
                  className="bg-bg-primary-fade text-text-maroon dark:bg-bg-maroon-secondary dark:text-text-secondary"
                >
                  <span>{t("Product.hot")}</span>
                </Badge>
              </>
            )
          )}
        </div>
      </div>

      {/* Product Information */}
      <div className={isPopular ? "space-y-2" : "space-y-1"}>
        <p
          className={cn(
            "font-primary truncate leading-none font-semibold",
            isPopular
              ? "text-text-maroon-700 text-lg"
              : "text-text-primary text-base",
          )}
        >
          <Link
            href={`/products/${product.id}`}
            className="focus-visible:after:outline-ring after:absolute after:inset-0 after:z-10 after:rounded-xl focus-visible:outline-none focus-visible:after:outline-2"

            {product.title}
          </Link>
        </p>

        {/* Rating & Cart */}
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-0.5">
            {Array.from({ length: starsCount }, (_, i) => (
              <Star
                key={i}
                size={isPopular ? 20 : 17}
                strokeWidth={1.5}
                fill={i < Math.round(product.rating ?? 0) ? "#F9A825" : "none"}
                className="text-[#F9A825]"
              />
            ))}
          </div>
          <div className="relative z-20">
            <AddToCartButton
              productId={String(product.id)}
              variant="icon"
              stock={product.stock}
            />
          </div>
        </div>

        {/* Price */}
        <div
          className={cn(
            "flex items-center gap-2",
            isPopular && "flex-wrap gap-1",
          )}
        >
          {hasDiscount && (
            <span
              className={cn(
                "font-primary text-text-muted align-bottom text-base leading-none font-medium tracking-normal line-through",
              )}
            >
              {product.price} {t("Product.currency")}
            </span>

              "font-primary align-bottom text-[14px] leading-none font-medium",

                : "text-bg-primary",

          >
            {discountedPrice} {t("Product.currency")}
          </span>
        </div>
      </div>
    </div>
  );
}

===== src/shared/components/rating-stars.tsx =====
import { Star } from "lucide-react";
const STAR_COUNT = 5;
const MAX_RATING = 5;
const ROUND_DOWN_MAX_FRACTION = 0.25;
const HALF_STAR_MAX_FRACTION = 0.75;

export type StarFill = 0 | 0.5 | 1;

export function getRatingStarFills(rating: number): StarFill[] {
  const emptyStars = Array.from({ length: STAR_COUNT }, () => 0 as StarFill);
  if (!Number.isFinite(rating)) return emptyStars;

  const clampedRating = Math.min(MAX_RATING, Math.max(0, rating));
  const wholeStars = Math.floor(clampedRating);
  const fraction = Number((clampedRating - wholeStars).toFixed(2));

  const displayedRating =
    fraction <= ROUND_DOWN_MAX_FRACTION
      ? wholeStars
      : fraction <= HALF_STAR_MAX_FRACTION
        ? wholeStars + 0.5
        : Math.min(wholeStars + 1, MAX_RATING);

  const fullStarCount = Math.floor(displayedRating);
  const hasHalfStar = displayedRating - fullStarCount === 0.5;

  return Array.from({ length: STAR_COUNT }, (_, index) => {
    if (index < fullStarCount) return 1;
    if (index === fullStarCount && hasHalfStar) return 0.5;
    return 0;
  });
}

export function RatingStars({ rating }: { rating: number }) {
  const starFills = getRatingStarFills(rating);

  return starFills.map((fill, index) => (
    <span key={index} className="relative inline-flex size-4">
      <Star size={16} strokeWidth={1.5} className="text-text-rating" />
      <span
        className="absolute inset-y-0 inset-s-0 overflow-hidden"
        style={{ width: `${fill * 100}%` }}
      >
        <Star
          size={16}
          strokeWidth={1.5}
          fill="currentColor"
          className="text-text-rating"
        />
      </span>
    </span>
  ));
}

              >
                <LayoutDashboard aria-hidden="true" className="size-4" />
                <span>{t("dashboard")}</span>
              </DropdownMenuItem>
            </DropdownMenuGroup>
          </>
        )}

        <FormDivider />
        <DropdownMenuGroup className="p-1.25">
          <DropdownMenuItem
            onClick={handleLogout}
            className="text-text-plain gap-2 px-2 py-1.5 text-sm font-medium"
          >
            <LogOut aria-hidden="true" className="size-4 rtl:rotate-180" />
            <span>{t("logout")}</span>
          </DropdownMenuItem>
        </DropdownMenuGroup>
      </DropdownMenuContent>
    </DropdownMenu>
  );
}
