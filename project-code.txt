"use client";

import React, { Component, ErrorInfo, ReactNode } from "react";

interface ErrorBoundaryProps {
  children: ReactNode;
  onError?: (error: Error, errorInfo: ErrorInfo) => void;
}

interface ErrorBoundaryState {
  hasError: boolean;
}

export class GracefullyDegradingErrorBoundary extends Component<
  ErrorBoundaryProps,
  ErrorBoundaryState
> {
  private contentRef: React.RefObject<HTMLDivElement | null>;

  constructor(props: ErrorBoundaryProps) {
    super(props);
    this.state = { hasError: false };
    this.contentRef = React.createRef();
  }

  static getDerivedStateFromError(): ErrorBoundaryState {
    return { hasError: true };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    if (this.props.onError) {
      this.props.onError(error, errorInfo);
    }
  }

  render() {
    if (this.state.hasError) {
      // Render the current HTML content without hydration
      return (
        <>
          <div
            ref={this.contentRef}
            suppressHydrationWarning
            dangerouslySetInnerHTML={{
              __html: this.contentRef.current?.innerHTML || "",
            }}
          />
          <div className="fixed right-0 bottom-0 left-0 bg-red-600 px-6 py-4 text-center text-white">
            <p className="font-semibold">
              An error occurred during page rendering
            </p>
          </div>
        </>
      );
    }

    return <div ref={this.contentRef}>{this.props.children}</div>;
  }
}

export default GracefullyDegradingErrorBoundary;

type State =
  | "loading"
  | "hidden" // السيرفر مش جاهز
  | "unsupported" // المتصفح مش بيدعم
  | "denied" // المستخدم رفض الإذن
  | "off"
  | "on"
  | "busy"
  | "error";

export function NotificationToggle() {
  const [state, setState] = useState<State>("loading");

  useEffect(() => {
    (async () => {
      if (!isPushSupported()) return setState("unsupported");
      const key = await getVapidPublicKey();
      if (!key) return setState("hidden");
      if (Notification.permission === "denied") return setState("denied");
      setState((await hasActiveSubscription()) ? "on" : "off");
    })();
  }, []);

  const handleEnable = async () => {
    setState("busy");
    try {
      const result = await enablePush();
      setState(result === "denied" ? "denied" : "on");
    } catch {
      setState("error");
    }
  };

  const handleDisable = async () => {
    setState("busy");
    try {
      await disablePush();
      setState("off");
    } catch {
      setState("error");
    }
  };

  if (state === "loading" || state === "hidden") return null;
  if (state === "unsupported") return <p>Notifications are not supported in this browser.</p>;
  if (state === "denied")
    return <p>Notifications are blocked. Enable them from your browser settings.</p>;

  return (
    <div>
      {state === "error" && <p>Something went wrong. Please try again.</p>}
      {state === "on" ? (
        <button onClick={handleDisable}>Disable notifications</button>
      ) : (
        <button onClick={handleEnable} disabled={state === "busy"}>
          {state === "busy" ? "Please wait..." : "Enable notifications"}
        </button>
      )}
    </div>
  );
}

const cookieMaxAge = 60 * 60 * 24 * 365;

function writeCookie(name: string, value: string) {
  document.cookie = `${name}=${value};path=/;max-age=${cookieMaxAge};samesite=lax`;
}

type ThemeContextValue = {
  theme: ThemeMode;
  resolvedTheme: ResolvedTheme;
  setTheme: (theme: ThemeMode) => void;
};

type ThemeState = {
  theme: ThemeMode;
  systemTheme: ResolvedTheme;
  initialized: boolean;
};

type ThemeAction =
  | { type: "initialize"; theme: ThemeMode; systemTheme: ResolvedTheme }
  | { type: "set-theme"; theme: ThemeMode }
  | { type: "system-change"; systemTheme: ResolvedTheme };

function themeReducer(state: ThemeState, action: ThemeAction): ThemeState {
  switch (action.type) {
    case "initialize":
      return {
        theme: action.theme,
        systemTheme: action.systemTheme,
        initialized: true,
      };
    case "set-theme":
      return { ...state, theme: action.theme };
    case "system-change":
      return { ...state, systemTheme: action.systemTheme };
  }
}

const ThemeContext = createContext<ThemeContextValue | null>(null);

export function ThemeProvider({
  children,
  initialTheme,
  initialSystemTheme,
}: {
  children: ReactNode;
  initialTheme: ThemeMode;
  initialSystemTheme: ResolvedTheme;
}) {
  // hooks
  // Seed from the server-read cookies so the rendered class matches <html>.
  const [state, dispatch] = useReducer(themeReducer, {
    theme: initialTheme,
    systemTheme: initialSystemTheme,
    initialized: true,
  });
  const resolvedTheme = resolveTheme(state.theme, state.systemTheme);

  // setters
  const setTheme = useCallback((theme: ThemeMode) => {
    const mode = parseTheme(theme);
    dispatch({ type: "set-theme", theme: mode });
    writeCookie(themeCookie, mode);
  }, []);

  // effects
  useLayoutEffect(() => {
    const media = window.matchMedia("(prefers-color-scheme: dark)");
    const systemTheme: ResolvedTheme = media.matches ? "dark" : "light";
    // Correct the server's (possibly stale) system value and persist it so the
    // next server render resolves "system" to the right class.
    if (systemTheme !== initialSystemTheme) {
      dispatch({ type: "system-change", systemTheme });
    }
    writeCookie(systemCookie, systemTheme);

    const onSystemChange = (event: MediaQueryListEvent) => {
      const next: ResolvedTheme = event.matches ? "dark" : "light";
      dispatch({ type: "system-change", systemTheme: next });
      writeCookie(systemCookie, next);
    };

    media.addEventListener("change", onSystemChange);
    return () => media.removeEventListener("change", onSystemChange);
  }, [initialSystemTheme]);

  useEffect(() => {
    const root = document.documentElement;
    root.classList.remove("light", "dark");
    root.classList.add(resolvedTheme);
  }, [resolvedTheme]);

  return (
    <ThemeContext value={{ theme: state.theme, resolvedTheme, setTheme }}>
      {children}
    </ThemeContext>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) throw new Error("useTheme must be used within a ThemeProvider");
  return context;
}

// Theme values shared by the server (layout reads cookies, sets the <html>
// class) and the client provider. Kept free of "use client" and React context
// so a Server Component can import it.

export const themeModes = ["light", "system", "dark"] as const;
export type ThemeMode = (typeof themeModes)[number];
export type ResolvedTheme = "light" | "dark";

// Cookies (not localStorage) so the server can read the theme during render
// and emit the class on <html>, avoiding a flash before hydration.
// themeCookie drives the toggle; systemCookie carries the client's last known
// OS preference so the server can resolve "system" to a concrete class.
export const themeCookie = "rose-theme";
export const systemCookie = "rose-system";

export function parseTheme(value: string | null | undefined): ThemeMode {
  return themeModes.includes(value as ThemeMode)
    ? (value as ThemeMode)
    : "system";
}

export function parseResolved(value: string | null | undefined): ResolvedTheme {
  return value === "dark" ? "dark" : "light";
}

export function resolveTheme(
  theme: ThemeMode,
  systemTheme: ResolvedTheme,
): ResolvedTheme {
  return theme === "system" ? systemTheme : theme;
}

  email: string,

import { BellOff, BrushCleaning, CheckCheck } from "lucide-react";

import { Button } from "@/components/ui/button";
import { cn } from "cn";
import { Card } from "@/components/ui/card";


export interface NotificationItem {
  id: string;
  title: string;
  read: boolean;
}

interface NotificationsPanelProps {
  notifications?: NotificationItem[];
  onClearAll?: () => void;
  onMarkAllAsRead?: () => void;
  className?: string;
}

const actionClass =
  "h-auto gap-2 px-1 py-1 text-lg font-medium text-zinc-400 hover:bg-transparent hover:text-zinc-600 [&_svg]:size-7";

export default function NotificationsPanel({
  notifications = [],
  onClearAll,
  onMarkAllAsRead,
  className,
}: NotificationsPanelProps) {
  const isEmpty = notifications.length === 0;

  return (
    <Card
      aria-label="Notifications"
      className={cn(
        "flex min-h-56 w-full max-w-84 flex-col gap-0 overflow-hidden  border-0 bg-white p-0 shadow-2xl",
        className
      )}
    >
      {/* Header */}
      <header className="bg-maroon-700 max-w-84 min-h-13 py-4 px-4">
        <h2 className="text-xl font-bold leading-tight text-white">
          Notifications
        </h2>
      </header>

      {/* Actions */}
      <div className="flex items-center justify-between border-b border-[#d9dadd] px-5 py-3">
        <Button variant="ghost" className={actionClass} onClick={onClearAll}>
          <BrushCleaning />
          Clear all notifications
        </Button>

        <Button
          variant="ghost"
          className={actionClass}
          onClick={onMarkAllAsRead}
        >
          <CheckCheck />
          Mark all as read
        </Button>
      </div>

      {/* Body */}
      <div className="flex flex-1 flex-col items-center justify-center gap-4 p-6">
        {isEmpty ? (
          <>
            <BellOff className="size-20 text-[#6b6f76]" strokeWidth={1.25} />
            <p className="text-xl text-[#6b6f76]">No notifications to display.</p>
          </>
        ) : (
          <ul className="w-full">
            {notifications.map((n) => (
              <li
                key={n.id}
                className={cn(
                  "border-b border-[#d9dadd] py-3",
                  n.read ? "font-normal" : "font-bold"
                )}
              >
                {n.title}
              </li>
            ))}
          </ul>
        )}
      </div>
    </Card>
  );
}

import { locale as getRootLocale } from "next/root-params";
import { hasLocale } from "next-intl";
import { getRequestConfig } from "next-intl/server";
import { routing } from "./routing";

export default getRequestConfig(async () => {
  const paramValue = await getRootLocale();

  let locale: string;
  if (paramValue && hasLocale(routing.locales, paramValue)) {
    locale = paramValue;
  } else {
    locale = routing.defaultLocale;
  }

  return {
    locale,
    messages: (await import(`./messages/${locale}.json`)).default,
  };
});

const API_URL =
  process.env.NEXT_PUBLIC_API_URL ?? "https://rose-app.elevate-bootcamp.cloud";

export async function sendSubscriptionToServer(subscription: PushSubscription) {
  const res = await fetch(`${API_URL}/api/notifications/subscriptions`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(subscription.toJSON()),
  });
  if (!res.ok) throw new Error("Failed to save subscription");
}

export async function deleteSubscriptionFromServer(endpoint: string) {
  const res = await fetch(`${API_URL}/api/notifications/subscriptions`, {
    method: "DELETE",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ endpoint }),
  });
  if (!res.ok) throw new Error("Failed to delete subscription");
}

const API_URL =
  process.env.NEXT_PUBLIC_API_URL ?? "https://rose-app.elevate-bootcamp.cloud";

export async function getVapidPublicKey(): Promise<string | null> {
  try {
    const res = await fetch(`${API_URL}/api/notifications/vapid-public-key`);

    if (res.status === 404 || !res.ok) return null;

    const data = await res.json();

    if (!data.pushConfigured) return null;

    return data.publicKey;
  } catch {
    return null;
  }
}

"use client";

import { getVapidPublicKey } from "./apis/notification-vapid-public-key";
import {
  deleteSubscriptionFromServer,
  sendSubscriptionToServer,
} from "./apis/notification-subscriptions";

function urlBase64ToUint8Array(base64String: string) {
  const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
  const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
  const rawData = atob(base64);
  return Uint8Array.from(rawData, (c) => c.charCodeAt(0));
}

export function isPushSupported() {
  return (
    typeof window !== "undefined" &&
    "serviceWorker" in navigator &&
    "PushManager" in window &&
    "Notification" in window
  );
}

export async function enablePush(): Promise<"subscribed" | "denied"> {
  const publicKey = await getVapidPublicKey();
  if (!publicKey) throw new Error("Push is not available");

  const permission = await Notification.requestPermission();
  if (permission !== "granted") return "denied";

  await navigator.serviceWorker.register("/sw.js");
  const registration = await navigator.serviceWorker.ready;

  const subscription =
    (await registration.pushManager.getSubscription()) ??
    (await registration.pushManager.subscribe({
      userVisibleOnly: true,
      applicationServerKey: urlBase64ToUint8Array(publicKey),
    }));

  await sendSubscriptionToServer(subscription);
  return "subscribed";
}

export async function disablePush() {
  const registration = await navigator.serviceWorker.getRegistration();
  const subscription = await registration?.pushManager.getSubscription();
  if (!subscription) return;

  const endpoint = subscription.endpoint;
  await subscription.unsubscribe();
  await deleteSubscriptionFromServer(endpoint);
}

export async function hasActiveSubscription() {
  const registration = await navigator.serviceWorker.getRegistration();
  return !!(await registration?.pushManager.getSubscription());
}

self.addEventListener("install", () => self.skipWaiting());
self.addEventListener("activate", (event) =>
  event.waitUntil(self.clients.claim()),
);

self.addEventListener("push", (event) => {
  let data = {};
  try {
    data = event.data ? event.data.json() : {};
  } catch {
    data = { message: event.data ? event.data.text() : "" };
  }

  const title = data.title || "Rose App";
  const options = {
    body: data.message || data.body || "",
    icon: data.icon || "/images/logo%201.png",
    data: { link: data.link || data.url || "/" },
  };

  event.waitUntil(self.registration.showNotification(title, options));
});

self.addEventListener("notificationclick", (event) => {
  event.notification.close();
  const url = new URL(
    event.notification.data?.link || "/",
    self.location.origin,
  ).href;

  event.waitUntil(
    (async () => {
      const windows = await self.clients.matchAll({
        type: "window",
        includeUncontrolled: true,
      });

      for (const client of windows) {
        if (new URL(client.url).origin === self.location.origin) {
          await client.focus();
          if ("navigate" in client) await client.navigate(url);
          return;
        }
      }

      await self.clients.openWindow(url);
    })(),
  );
});
